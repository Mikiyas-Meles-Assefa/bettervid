  const wp = [[-80,45],[-83,42],[-86,39],[-89,37],[-93,35],[-96,32],[-98,29],[-99.5,25],[-100,22],[-99.5,19.5]];
  const rt = [];
  for (let i = 0; i < wp.length - 1; i++) {
    for (let k = 0; k < 8; k++) {
      const t = k / 8;
      rt.push([lerp(wp[i][0], wp[i + 1][0], t), lerp(wp[i][1], wp[i + 1][1], t)]);
    }
  }
  rt.push(wp[wp.length - 1]);
  const start = rt[0], end = rt[rt.length - 1], mid = rt[Math.floor(rt.length / 2)];
  path("s1_map", { x: 0, y: 0, points: flat(outline), closed: 1, fill: "strip", stroke: "faded", strokeWidth: 4 });
  path("s1_route", { x: 0, y: 0, points: flat(rt), closed: 0, fill: "none", stroke: "accent", strokeWidth: 10, drawn: 0 });

  path("s1_start", { x: px(start[0]), y: py(start[1]), points: circle(0, 0, 14), fill: "ink" });
  path("s1_mid", { x: px(mid[0]), y: py(mid[1]), points: circle(0, 0, 6), fill: "ink" });
  path("s1_end", { x: px(end[0]), y: py(end[1]), points: circle(0, 0, 14), fill: "ink" });
  const ga = wf("Their"), gb = wf("trip");
  for (let i = 0; i < 5; i++) {
    const p = rt[Math.round((0.2 + i * 0.15) * (rt.length - 1))];
    const id = "s1_gen" + i;
    path(id, { x: px(p[0]), y: py(p[1]), points: circle(0, 0, 9), fill: "warm", opacity: 0 });
    key(id, "opacity", lerp(ga, gb, i / 5), 1, "smooth");
  }
end
