  }
  for (let i = 97; i < 99; i++) {

    const pts = [];
    for (let k = 0; k <= 40; k++) {

      pts.push(Math.cos(a) * R, Math.sin(a) * R);
    }
    const u = path("s1_under", { x: c.x, y: c.y, points: pts, closed: 1, fill: "tubeLow", opacity: 0 });

  }

    key(sl, "opacity", Math.round(args.sl + 4), 1, "smooth");
  }
