  for (let i = 0; i < args.n; i++) {

program s1_lattice rows=3 cols=3 done=0 doc="Molecules locking into an open hexagonal lattice, one bond at a time."
  const R = Math.min(W / ((args.cols + 0.5) * 1.732), H / ((args.rows - 1) * 1.5 + 2)) * 0.9;
  const vmap = {}, verts = [], edges = [], ek = {};
  const vid = (x, y) => {
    const k = Math.round(x * 10) + "_" + Math.round(y * 10);
    if (!(k in vmap)) { vmap[k] = verts.length; verts.push([x, y]); }
    return vmap[k];
  };
  for (let r = 0; r < args.rows; r++) for (let c = 0; c < args.cols; c++) {
    const cx = R * 1.732 * (c + 0.5 * (r % 2)), cy = R * 1.5 * r;
    const ids = [];
    for (let k = 0; k < 6; k++) { const a = Math.PI / 6 + k * Math.PI / 3; ids.push(vid(cx + R * Math.cos(a), cy + R * Math.sin(a))); }
    for (let k = 0; k < 6; k++) {
      const p = ids[k], q = ids[(k + 1) % 6];
      const pair = Math.min(p, q) + "-" + Math.max(p, q);
      if (!(pair in ek)) { ek[pair] = 1; edges.push([p, q]); }
    }
  }
  let x0 = 1e9, x1 = -1e9, y0 = 1e9, y1 = -1e9;
  for (const v of verts) { x0 = Math.min(x0, v[0]); x1 = Math.max(x1, v[0]); y0 = Math.min(y0, v[1]); y1 = Math.max(y1, v[1]); }
  const ox = W / 2 - (x0 + x1) / 2, oy = H / 2 - (y0 + y1) / 2;
  const cxm = (x0 + x1) / 2, cym = (y0 + y1) / 2;
  const edist = (e) => (Math.hypot(verts[e[0]][0] - cxm, verts[e[0]][1] - cym) + Math.hypot(verts[e[1]][0] - cxm, verts[e[1]][1] - cym)) / 2;
  edges.sort((a, b) => edist(a) - edist(b));
  const V = verts.length, E = edges.length;
  for (let i = 0; i < V; i++) {
    const px = verts[i][0] + ox, py = verts[i][1] + oy;
    const id = path("a" + i, { x: px, y: py, points: circle(0, 0, R * 0.22), fill: "ink", opacity: 0 });
    if (args.done) key(id, "opacity", T0, 1);
    else {
      key(id, "opacity", T0, 0);
      key(id, "opacity", lerp(T0, T1, 0.5 * i / V), 1, "smooth");
    }
  }
  edges.forEach((e, j) => {
    const a = verts[e[0]], b = verts[e[1]];
    const id = path("b" + j, { points: [a[0] + ox, a[1] + oy, b[0] + ox, b[1] + oy], stroke: "accent", strokeWidth: 7, drawn: 0 });
    if (args.done) key(id, "drawn", T0, 1);
    else {
      const s = lerp(T0, T1, 0.9 * j / E);
      key(id, "drawn", T0, 0);
      key(id, "drawn", s, 0);
      key(id, "drawn", s + 8, 1, "smooth");
    }
  });
end
