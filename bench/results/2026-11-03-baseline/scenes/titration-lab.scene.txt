program drops n=12 fall=12 doc="Drops leave the burette and fall into the flask, one after another."
  for (let i = 0; i < args.n; i++) {
    const s = lerp(T0, T1, i / args.n);
    const bx = on.buret.x, by = on.buret.y + on.buret.h / 2, fy = on.flask.y;
    const id = path("d" + i, { x: bx, y: by, points: circle(0, 0, 9), fill: "accent", opacity: 0 });
    key(id, "opacity", T0, 0);
    key(id, "opacity", s, 1);
    anim(id, "y", s, s + args.fall, (t) => by + (fy - by) * t * t, 1);
    key(id, "opacity", s + args.fall, 0);
  }
end
