program s1_cells n=36 doc="B cells in a field; three match the piece and multiply, one copy of each stays as memory."
  const cols = 9, rows = 4;
  const cw = W / cols, ch = H / rows, rad = Math.min(cw, ch) * 0.34;
  const match = [4, 13, 22];
  const mid = (i) => { const c = i % cols, r = Math.floor(i / cols); return [(c + 0.5) * cw, (r + 0.5) * ch]; };
  for (let i = 0; i < args.n; i++) {
    const [x, y] = mid(i);
    const isM = match.includes(i);
    const id = path("s1_b" + i, { x, y, points: circle(0, 0, rad), closed: 1, fill: isM ? "accent" : "soft", opacity: 0 });
    key(id, "opacity", T0, 0);
    key(id, "opacity", lerp(T0, T1, 0.4 * i / args.n), 1, "smooth");
    key(id, "opacity", lerp(T0, T1, 0.97), 0.25, "smooth");
    if (isM) {
      key(id, "scaleX", lerp(T0, T1, 0.45), 1.3, "smooth");
      key(id, "scaleY", lerp(T0, T1, 0.45), 1.3, "smooth");
      key(id, "scaleX", lerp(T0, T1, 0.6), 1, "smooth");
      key(id, "scaleY", lerp(T0, T1, 0.6), 1, "smooth");
    }
  }
  for (const i of match) {
    const [x, y] = mid(i);
    const mem = path("s1_k" + i + "_0", { x, y: y - ch * 0.3, points: circle(0, 0, rad * 0.9), closed: 1, fill: "accent", opacity: 0 });
    key(mem, "opacity", T0, 0);
    key(mem, "opacity", lerp(T0, T1, 0.5), 1, "smooth");
    const eff = path("s1_k" + i + "_1", { x, y: y + ch * 0.3, points: circle(0, 0, rad * 0.9), closed: 1, fill: "accent", opacity: 0 });
    key(eff, "opacity", T0, 0);
    key(eff, "opacity", lerp(T0, T1, 0.56), 1, "smooth");
    key(eff, "opacity", lerp(T0, T1, 0.9), 0, "smooth");
  }
end
