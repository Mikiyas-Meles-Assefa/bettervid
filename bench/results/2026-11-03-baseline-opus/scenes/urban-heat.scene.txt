  const base = H - 90, scale = (H - 220) / 55;
  const bw = 140, ax = W * 0.25, cx = W * 0.75;
  path("axis", { x: 0, y: 0, points: [30, base, W - 20, base], stroke: "ink", strokeWidth: 5 });
  const bars = [
    { id: "asphalt", name: "asphalt", v: 52, x: ax, at: args.asphaltAt, val: args.asphaltVal, fill: "slate" },
    { id: "canopy", name: "tree canopy", v: 29, x: cx, at: args.canopyAt, val: args.canopyVal, fill: "teal" }
  ];
  bars.forEach((b) => {
    const h = b.v * scale;
    const p = path(b.id, { x: b.x, y: base, points: rect(-bw / 2, -h, bw, h), closed: 1, fill: b.fill, scaleY: 0.001 });
    key(p, "scaleY", b.at, 0.001);
    key(p, "scaleY", b.at + 22, 1, "out");
    const nm = text(b.id + "_name", { x: b.x, y: base + 48, text: b.name, size: 42, color: "ink", opacity: 0 });
    key(nm, "opacity", b.at, 0);
    key(nm, "opacity", b.at + 10, 1, "smooth");
    const v = text(b.id + "_val", { x: b.x, y: base - h - 40, text: b.v + " °C", size: 58, weight: 700, color: "ink", opacity: 0 });
    key(v, "opacity", b.val, 0);
    key(v, "opacity", b.val + 8, 1, "smooth");
  });
  const yA = base - 52 * scale, yC = base - 29 * scale;
  const band = path("gap", { x: 0, y: 0, points: rect(ax + bw / 2, yA, cx - ax - bw, yC - yA), closed: 1, fill: "orangeDeep", opacity: 0 });
  key(band, "opacity", args.gapAt, 0);
  key(band, "opacity", args.gapAt + 12, 0.3, "smooth");
  const lvl = path("level", { x: 0, y: 0, points: [ax + bw / 2, yA, cx + bw / 2, yA], stroke: "orangeDeep", strokeWidth: 6, drawn: 0 });
  key(lvl, "drawn", args.gapAt, 0);
  key(lvl, "drawn", args.gapAt + 14, 1, "smooth");
  const ar = path("gaparrow", { x: 0, y: 0, points: [ax + bw / 2 + 24, yA + 8, ax + bw / 2 + 24, yC], stroke: "orangeDeep", strokeWidth: 8, drawn: 0 });
  key(ar, "drawn", args.gapAt + 6, 0);
  key(ar, "drawn", args.gapAt + 20, 1, "smooth");
  const gv = text("gap_val", { x: (ax + cx) / 2 + 18, y: (yA + yC) / 2, text: "23 °C", size: 52, weight: 700, color: "orangeDeep", opacity: 0 });
  key(gv, "opacity", args.gapVal, 0);
  key(gv, "opacity", args.gapVal + 8, 1, "smooth");
end
